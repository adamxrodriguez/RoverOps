cmake_minimum_required(VERSION 3.24)
project(RoverOps VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
include(CTest)

option(ROVEROPS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ROVEROPS_TSAN "Enable ThreadSanitizer (separate build from ASan)" OFF)
if(ROVEROPS_SANITIZE AND ROVEROPS_TSAN)
  message(FATAL_ERROR "Use separate builds for ROVEROPS_SANITIZE and ROVEROPS_TSAN")
endif()

add_library(roverops_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(roverops_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(ROVEROPS_SANITIZE)
    target_compile_options(roverops_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(roverops_options INTERFACE -fsanitize=address,undefined)
  elseif(ROVEROPS_TSAN)
    target_compile_options(roverops_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(roverops_options INTERFACE -fsanitize=thread)
  endif()
endif()

add_library(roverops_json INTERFACE)
target_include_directories(roverops_json SYSTEM INTERFACE third_party/json)

add_library(roverops_core core/src/rover.cpp)
target_include_directories(roverops_core PUBLIC core/include)
target_link_libraries(roverops_core PUBLIC roverops_options)

add_library(roverops_service service/src/simulation.cpp service/src/protocol.cpp)
target_include_directories(roverops_service PUBLIC service/include)
target_link_libraries(roverops_service PUBLIC roverops_core roverops_json Threads::Threads)

add_executable(roverops service/src/main.cpp service/src/api_server.cpp)
target_include_directories(roverops SYSTEM PRIVATE third_party/httplib)
target_compile_definitions(roverops PRIVATE CPPHTTPLIB_WEBSOCKET_MAX_PAYLOAD_LENGTH=4096 CPPHTTPLIB_WEBSOCKET_CLOSE_TIMEOUT_SECOND=1)
target_link_libraries(roverops PRIVATE roverops_service)

if(BUILD_TESTING)
  add_executable(roverops_tests tests/rover_tests.cpp tests/service_tests.cpp)
  target_include_directories(roverops_tests SYSTEM PRIVATE third_party/doctest)
  target_link_libraries(roverops_tests PRIVATE roverops_service)
  add_test(NAME roverops_unit COMMAND roverops_tests)
  set_tests_properties(roverops_unit PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 30)
endif()
